# 全域統計變數
total_games = 0
wins = 0

# 遊戲狀態儲存: {game_id: {"target": str, "status": str, "hard_mode": bool, "guesses": list}}
games_db: Dict[str, dict] = {}


class NewGameRequest(BaseModel):
    hard_mode: bool = False


class GuessRequest(BaseModel):
    game_id: str
    guess: str


@app.get("/api/stats")
def get_stats():
    """ 取得總遊玩局數與勝率資料 """
    win_rate = (wins / total_games * 100) if total_games > 0 else 0.0
    return {
        "total_games": total_games,
        "wins": wins,
        "win_rate": round(win_rate, 2)
    }


@app.post("/api/new-game")
def start_new_game(req: Optional[NewGameRequest] = None):
    """ 建立新一局遊戲，支援 hard_mode 參數 """
    global total_games
    total_games += 1

    hard_mode = req.hard_mode if req else False
    game_id = str(uuid.uuid4())
    target_word = random.choice(WORD_BANK)
    
    games_db[game_id] = {
        "target": target_word,
        "status": "PLAYING",
        "hard_mode": hard_mode,
        "guesses": []  # 記錄已猜過的歷史與正確綠色位置
    }
    return {
        "game_id": game_id, 
        "message": "New game started!",
        "hard_mode": hard_mode,
        "total_games": total_games,
        "wins": wins
    }


@app.post("/api/guess")
def process_guess(req: GuessRequest):
    global wins
    game_id = req.game_id
    guess = req.guess.upper()

    if game_id not in games_db:
        raise HTTPException(status_code=404, detail="Game session not found.")
    
    if len(guess) != 5:
        raise HTTPException(status_code=400, detail="Word must be 5 letters.")

    # 任務二：字典檢驗
    if guess not in WORD_BANK:
        raise HTTPException(status_code=400, detail="Not in word list")

    game_data = games_db[game_id]
    target = game_data["target"]
    hard_mode = game_data["hard_mode"]
    past_guesses = game_data["guesses"]

    # 任務四：Hard Mode 邏輯驗證
    if hard_mode and past_guesses:
        # 檢查是否違反了先前確定的綠色（Correct）字母與位置
        last_guess_info = past_guesses[-1]
        last_guess_word = last_guess_info["word"]
        last_feedback = last_guess_info["feedback"]

        for i in range(5):
            if last_feedback[i] == "correct":
                # 如果前一次這裡黃金綠字是某個字母，這次猜測該位置必須一模一樣
                required_char = last_guess_word[i]
                if guess[i] != required_char:
                    raise HTTPException(
                        status_code=400, 
                        detail="Hard Mode: Must use revealed hint at green position"
                    )

    feedback = ["absent"] * 5
    target_letter_counts = {}

    for char in target:
        target_letter_counts[char] = target_letter_counts.get(char, 0) + 1

    for i in range(5):
        if guess[i] == target[i]:
            feedback[i] = "correct"
            target_letter_counts[guess[i]] -= 1

    for i in range(5):
        if feedback[i] != "correct" and guess[i] in target:
            if target_letter_counts[guess[i]] > 0:
                feedback[i] = "present"
                target_letter_counts[guess[i]] -= 1

    # 紀錄本次猜測到歷史中
    past_guesses.append({"word": guess, "feedback": feedback})

    is_correct = (guess == target)
    game_over = False
    
    if is_correct:
        game_data["status"] = "WON"
        wins += 1
        game_over = True
    else:
        if len(past_guesses) >= 6:
            game_data["status"] = "LOST"
            game_over = True

    # 計算勝率
    win_rate = (wins / total_games * 100) if total_games > 0 else 0.0

    # 任務一：若遊戲結束（或猜對、或滿 6 次失敗），回傳 target_word
    response_target = target if (is_correct or len(past_guesses) == 6) else None

    return {
        "guess": guess,
        "feedback": feedback,
        "is_correct": is_correct,
        "game_over": game_over,
        "target_word": response_target,
        "total_games": total_games,
        "wins": wins,
        "win_rate": round(win_rate, 2)
    }
